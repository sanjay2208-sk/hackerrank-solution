#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n, q;
    // n: number of arrays, q: number of queries
    cin >> n >> q;

    // A vector of vectors to create the variable-sized 2D structure
    vector<vector<int>> a(n);

    for (int i = 0; i < n; i++) {
        int k;
        cin >> k; // Size of the current array
        a[i].resize(k);
        for (int j = 0; j < k; j++) {
            cin >> a[i][j];
        }
    }

    // Handle each query
    for (int count = 0; count < q; count++) {
        int i, j;
        cin >> i >> j;
        // Output the element at the specified index
        cout << a[i][j] << endl;
    }

    return 0;
}
