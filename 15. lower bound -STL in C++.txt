#include <iostream>
#include <vector>
#include <algorithm> // Required for lower_bound

using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);
    for (int i = 0; i < n; i++) {
        cin >> v[i];
    }

    int q;
    cin >> q;
    while (q--) {
        int query;
        cin >> query;
        
        // lower_bound returns an iterator to the first element 
        // that is NOT less than the query.
        auto it = lower_bound(v.begin(), v.end(), query);
        
        // Convert iterator to 1-based index
        int index = (it - v.begin()) + 1;

        if (it != v.end() && *it == query) {
            cout << "Yes " << index << endl;
        } else {
            cout << "No " << index << endl;
        }
    }
    return 0;
}
