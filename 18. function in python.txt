def is_leap(year):
    leap = False
    
    # A year is a leap year if:
    # 1. It is divisible by 4
    # 2. BUT if it is divisible by 100, it must also be divisible by 400
    if (year % 4 == 0):
        if (year % 100 == 0):
            if (year % 400 == 0):
                leap = True
            else:
                leap = False
        else:
            leap = True
            
    return leap

# The template handles input/output
year = int(input())
print(is_leap(year))
