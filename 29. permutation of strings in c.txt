#include <stdio.h>

int main() {
    int n, sum = 0;
    
    // Read the five-digit integer
    scanf("%d", &n);
    
    // Iterate until all digits are processed
    while (n > 0) {
        sum += (n % 10); // Extract the last digit and add to sum
        n /= 10;         // Remove the last digit
    }
    
    printf("%d\n", sum);
    
    return 0;
}
