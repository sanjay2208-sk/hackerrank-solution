#include <iostream>
#include <vector>

using namespace std;

int main() {
    int n;
    // Read the size of the array
    if (cin >> n) {
        // Use a vector or array to store elements
        vector<int> arr(n);
        
        // Input n integers
        for (int i = 0; i < n; i++) {
            cin >> arr[i];
        }
        
        // Print elements in reverse order
        for (int i = n - 1; i >= 0; i--) {
            cout << arr[i];
            if (i > 0) {
                cout << " "; // Space between numbers
            }
        }
        cout << endl;
    }
    
    return 0;
}
