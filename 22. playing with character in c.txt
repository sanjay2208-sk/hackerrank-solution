#include <stdio.h>

int main() {
    char ch;
    char s[100];
    char sen[100];

    // Read a single character
    scanf("%c", &ch);
    
    // Read a word
    scanf("%s", s);
    
    // Clear the buffer and read a full sentence (including spaces)
    scanf("\n%[^\n]%*c", sen);

    // Print the results
    printf("%c\n", ch);
    printf("%s\n", s);
    printf("%s\n", sen);

    return 0;
}
